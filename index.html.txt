    <script>
        // База демонстрационных вопросов для СОР
        const mockQuestions = [
            {
                id: 1,
                type: "radio",
                text: "1. Чему равен результат выражения 5 + 3 * 2?",
                options: ["16", "11", "13", "10"],
                correct: "11"
            },
            {
                id: 2,
                type: "radio",
                text: "2. Какая структура данных работает по принципу FIFO (первым вошел — первым вышел)?",
                options: ["Стек (Stack)", "Очередь (Queue)", "Дерево (Tree)", "Граф (Graph)"],
                correct: "Очередь (Queue)"
            },
            {
                id: 3,
                type: "textarea",
                text: "3. Опишите своими словами основную цель создания данного СОР (развернутый ответ):",
                correct: "" // Текстовые вопросы обычно проверяются учителем вручную, здесь оценим по наличию текста
            }
        ];

        let studentData = {};
        let timerInterval;
        let timeLeft = 20 * 60; // 20 минут в секундах

        function startSOR(event) {
            event.preventDefault();
            
            // Сохраняем данные ученика
            studentData.fullname = document.getElementById('fullname').value;
            studentData.grade = document.getElementById('grade').value;
            studentData.subject = document.getElementById('subject').value;

            // Переключаем экраны
            document.getElementById('auth-screen').classList.add('hidden');
            document.getElementById('sor-screen').classList.remove('hidden');

            // Заполняем шапку СОР
            document.getElementById('sor-title').innerText = `СОР по предмету: ${studentData.subject}`;
            document.getElementById('student-info').innerText = `Ученик: ${studentData.fullname} | ${studentData.grade} класс`;

            // Рендерим вопросы
            renderQuestions();

            // Запускаем таймер
            startTimer();
        }

        function renderQuestions() {
            const container = document.getElementById('questions-container');
            container.innerHTML = '';

            mockQuestions.forEach((q, index) => {
                let qDiv = document.createElement('div');
                qDiv.className = 'question-card';

                let html = `<p>${q.text}</p>`;

                if (q.type === 'radio') {
                    html += `<div class="options">`;
                    q.options.forEach(opt => {
                        html += `
                            <label>
                                <input type="radio" name="q_${q.id}" value="${opt}" required>
                                ${opt}
                            </label>
                        `;
                    });
                    html += `</div>`;
                } else if (q.type === 'textarea') {
                    html += `<textarea name="q_${q.id}" placeholder="Введите ваш развернутый ответ здесь..." required></textarea>`;
                }

                qDiv.innerHTML = html;
                container.appendChild(qDiv);
            });
        }

        function startTimer() {
            const timerDisplay = document.getElementById('timer');
            
            timerInterval = setInterval(() => {
                let minutes = Math.floor(timeLeft / 60);
                let seconds = timeLeft % 60;

                timerDisplay.textContent = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;

                if (timeLeft <= 0) {
                    clearInterval(timerInterval);
                    alert("Время вышло! СОР будет автоматически отправлен.");
                    document.getElementById('sor-form').requestSubmit();
                }
                timeLeft--;
            }, 1000);
        }

        function submitSOR(event) {
            event.preventDefault();
            clearInterval(timerInterval);

            // Сбор ответов и проверка
            let score = 0;
            let autoCheckableCount = 0;

            mockQuestions.forEach(q => {
                if (q.type === 'radio') {
                    autoCheckableCount++;
                    const selectedOpt = document.querySelector(`input[name="q_${q.id}"]:checked`);
                    if (selectedOpt && selectedOpt.value === q.correct) {
                        score++;
                    }
                } else if (q.type === 'textarea') {
                    // Текстовый вопрос оценивается в 1 балл, если ученик что-то написал
                    autoCheckableCount++;
                    const textVal = document.querySelector(`textarea[name="q_${q.id}"]`).value.trim();
                    if (textVal.length > 5) {
                        score++; // Условный балл за заполненный развернутый ответ
                    }
                }
            });

            // Переключаем на экран результатов
            document.getElementById('sor-screen').classList.add('hidden');
            document.getElementById('result-screen').classList.remove('hidden');

            document.getElementById('result-student-details').innerText = `${studentData.fullname}, ${studentData.grade} класс (${studentData.subject})`;
            document.getElementById('score-display').innerText = `${score} / ${autoCheckableCount}`;
            
            if (score === autoCheckableCount) {
                document.getElementById('result-message').innerText = "Отличный результат! Вы справились со всеми заданиями.";
            } else {
                document.getElementById('result-message').innerText = "Работа сдана успешно. Ваши ответы отправлены преподавателю.";
            }
        }
    </script>
